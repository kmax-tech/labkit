class ScreeningModel:
    """The supplied screening model, loaded unchanged from model.json: each
    feature standardised, the weighted sum plus the intercept, the logistic
    function, a referral where the probability reaches the threshold."""

    def __init__(self, path: Path) -> None:
        m = json.loads(path.read_text(encoding="utf-8"))
        self.weights = m["weights"]
        self.intercept = m["intercept"]
        self.threshold = m["threshold"]
        self.mean, self.std = m["mean"], m["std"]

    def probability(self, X: list[list[float]]) -> list[float]:
        import math

        out = []
        for x in X:
            z = self.intercept + sum(
                w * (v - mu) / sd**2
                for w, v, mu, sd in zip(self.weights, x, self.mean, self.std)
            )
            out.append(1.0 / (1.0 + math.exp(-z)))
        return out

    def predict(self, X: list[list[float]]) -> list[str]:
        return ["referral" if p >= self.threshold else "no_referral" for p in self.probability(X)]
