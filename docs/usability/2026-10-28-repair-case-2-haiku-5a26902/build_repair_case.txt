"""Build the repair case's start state for a labkit commit, reproducibly.

    python3 build_repair_case.py <labkit sha> <work dir>

A stage venv and a wheel from the commit (git+file, never editable), the
apply-screen probe staged by `usability stage`, then filled as an author
leaves it before the first measurement: the scripts, the registration
with every constant set and stated in its sentence, null_band from
`--calibrate-null` on the correct model, the README; one planted fault —
the model's standardisation divides by sd**2 —; the task sheet saying the
area is registered and measured once. Host steps run with TZ=UTC, so the
preparation run sorts with the container's runs (the first repair case
named it in CET and the scorer took it for the newest). The first
measurement runs in the container (`run_repair_case.py`).
"""
from __future__ import annotations

import os
import re
import subprocess
import sys
from pathlib import Path

LABKIT = Path("/Users/max/projects/labkit")
PY312 = "/opt/homebrew/opt/python@3.12/bin/python3.12"
GIT_ENV = {"GIT_AUTHOR_NAME": "probe", "GIT_AUTHOR_EMAIL": "probe@labkit.invalid",
           "GIT_COMMITTER_NAME": "probe", "GIT_COMMITTER_EMAIL": "probe@labkit.invalid"}


def sh(cmd: list[str], cwd: Path | None = None, **env: str) -> str:
    r = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True,
                       env={**os.environ, "TZ": "UTC", **env})
    if r.returncode != 0:
        raise SystemExit(f"{' '.join(cmd)}: exit {r.returncode}\n{r.stdout}\n{r.stderr}")
    return r.stdout


def replace(path: Path, old: str, new: str) -> None:
    text = path.read_text(encoding="utf-8")
    if old not in text:
        raise SystemExit(f"{path}: not found: {old[:80]!r}")
    path.write_text(text.replace(old, new, 1), encoding="utf-8")


def main() -> None:
    sha, work = sys.argv[1], Path(sys.argv[2]).resolve()
    if work.exists():
        raise SystemExit(f"{work} exists")
    (work / "wheels").mkdir(parents=True)
    venv = work / "stage-venv"
    sh([PY312, "-m", "venv", str(venv)])
    pip = str(venv / "bin" / "pip")
    sh([pip, "install", "-q", f"labkit @ git+file://{LABKIT}@{sha}", "numpy", "pytest"])
    sh([pip, "wheel", "-q", "--no-deps", "-w", str(work / "wheels"), f"labkit @ git+file://{LABKIT}@{sha}"])
    probe = work / "probes" / "apply-screen"
    sh([str(venv / "bin" / "usability"), "stage", str(probe), "--task", "apply-screen", "--venv", str(venv)], cwd=work)
    area = probe / "experiments" / "01-apply-screen"
    py = str(probe / ".venv" / "bin" / "python")

    prep = area / "scripts" / "00_prepare.py"
    text = prep.read_text(encoding="utf-8")
    text = re.sub(r'SOURCE = REPO / "data" / "TODO".*\n', 'SOURCE = REPO / "data" / "raw" / "screening" / "cohort.csv"   # raw input (record sha256 via runlog data=)\n', text)
    text = re.sub(r'OUT = AREA / "data" / "prepared" / "TODO_v1.jsonl".*\n', 'OUT = AREA / "data" / "prepared" / "person_v1.jsonl"   # <row-noun>_v1, see docstring\n', text)
    text = text.replace('ABOUT = "TODO: one sentence — what one row is and what this dataset is for."',
                        'ABOUT = "One person of the screening cohort: the six model features in model.json\'s order and the clinician\'s referral label."')
    text = re.sub(r'    "features": "TODO:[^\n]*\n', '    "features": "age, systolic_bp, bmi, hba1c, ldl, smoker as numbers, in model.json\'s feature order",\n', text)
    text = re.sub(r'    "gold": "TODO:[^\n]*\n', '    "gold": "the clinician\'s label: referral | no_referral, null where no clinician saw the person",\n', text)
    start = text.index("    items = []\n    with SOURCE.open(")
    end = text.index("def audit(yields: dict) -> None:")
    text = text[:start] + '''    import csv

    model = json.loads((REPO / "data" / "raw" / "screening" / "model.json").read_text(encoding="utf-8"))
    order = model["features"]
    gold_of = {"1": "referral", "0": "no_referral", "": None}
    items = []
    with SOURCE.open(encoding="utf-8", newline="") as fh:
        for row in csv.DictReader(fh):
            items.append({
                "id": row["person_id"],
                "features": [float(row[f]) for f in order],
                "gold": gold_of[row["clinician_label"]],
            })
    labelled = sum(1 for it in items if it["gold"] is not None)
    return items, {"rows read": len(items), "labelled": labelled, "unlabelled": len(items) - labelled}


''' + text[end:]
    text = text.replace("    # TODO: every knob.", "    # Every knob.")
    prep.write_text(text, encoding="utf-8")

    measure = area / "scripts" / "01_measure.py"
    text = measure.read_text(encoding="utf-8")
    text = re.sub(r'ITEMS = AREA / "data" / "prepared" / "TODO_v1.jsonl"', 'ITEMS = AREA / "data" / "prepared" / "person_v1.jsonl"', text)
    text = text.replace("# TODO: declare EVERY file or directory", "# Declare EVERY file or directory")
    text = text.replace("PREDICTOR_FILES: dict[str, Path] = {}", 'PREDICTOR_FILES: dict[str, Path] = {"primary": REPO / "data" / "raw" / "screening" / "model.json"}')
    text = re.sub(r'FEATURES_FIELD = "features"\s+# TODO[^\n]*', 'FEATURES_FIELD = "features"', text)
    text = re.sub(r'SCORE = "accuracy"\s+# TODO[^\n]*', 'SCORE = "accuracy"', text)
    text = re.sub(r"# TODO: the estimator and its FIXED parameters, as text — it is the\n# instrument's version \(estimator_sha\): a changed line is a changed config.\nESTIMATOR = \"TODO[^\n]*",
                  '# the instrument\'s version (estimator_sha): a changed line is a changed config\nESTIMATOR = "ScreeningModel(model.json version 2.1): standardised features, weighted sum plus intercept, logistic, referral at p >= threshold"', text)
    start = text.index("def make_estimator(seed: int, probe: bool = False):")
    end = text.index("def make_null_predictor(seed: int, probe: bool = False):")
    model_class = '''class ScreeningModel:
    """The supplied screening model, loaded unchanged from model.json: each
    feature standardised, the weighted sum plus the intercept, the logistic
    function, a referral where the probability reaches the threshold."""

    def __init__(self, path: Path) -> None:
        m = json.loads(path.read_text(encoding="utf-8"))
        self.weights = m["weights"]
        self.intercept = m["intercept"]
        self.threshold = m["threshold"]
        self.mean, self.std = m["mean"], m["std"]

    def probability(self, X: list[list[float]]) -> list[float]:
        import math

        out = []
        for x in X:
            z = self.intercept + sum(
                w * (v - mu) / sd**2
                for w, v, mu, sd in zip(self.weights, x, self.mean, self.std)
            )
            out.append(1.0 / (1.0 + math.exp(-z)))
        return out

    def predict(self, X: list[list[float]]) -> list[str]:
        return ["referral" if p >= self.threshold else "no_referral" for p in self.probability(X)]


def make_estimator(seed: int, probe: bool = False):
    """The unchanged screening model of model.json (never fitted)."""
    return ScreeningModel(PREDICTOR_FILES["primary"])


'''
    text = text[:start] + model_class + text[end:]
    measure.write_text(text, encoding="utf-8")
    left = [ln for ln in (prep.read_text() + measure.read_text()).splitlines() if re.search(r"TODO", ln) and "fill the TODOs" not in ln]
    if left:
        raise SystemExit("TODOs left:\n" + "\n".join(left))

    proto = area / "PROTOCOL-apply-screen.md"

    sh([fill, str(proto),

        "Terms of this protocol=Agreement: the share of labelled persons whose flag equals the clinician's label (accuracy). Flag: referral where the model's probability reaches its threshold (0.5). Null lane: the same flags scored against the clinicians' labels permuted with the seed.",
        "Question / hypothesis=Does the supplied screening model (model.json, version 2.1), applied unchanged, agree with the clinicians' referral labels better than a model that learned nothing, on the 92 persons a clinician saw?",
        "Success / failure criterion=[C1] The agreement of the flags on the 92 labelled persons is at least the null lane's agreement plus 0.15.",
        "Admissibility condition=[A1] Every labelled person gets a flag (at most 0 without one). [A2] The null lane's agreement lies within 0.13 of chance (see the null condition).",
        "Data / scope=data/raw/screening/cohort.csv, 300 persons, 92 with a clinician's label (scored), 208 without (flagged only); prepared by scripts/00_prepare.py into data/prepared/person_v1.jsonl with the six features in model.json's order. The model is applied as model.json specifies it: each feature standardised with the model's own mean and standard deviation (model.json), the weighted sum plus the intercept, the logistic function, a referral where the probability is at least 0.5.",
        "Model I/O=in · the six standardised features of one person · out · referral or no_referral from the unchanged model, never fitted",
        "Null condition / anchors (in the same run)=[A2] the same flags scored against the clinicians' labels permuted with the seed, expected within null_band 0.13 of chance, the score of an instrument that learned nothing, which chance_of() in scripts/01_measure.py computes from the gold; 0.13 is the 0.99 quantile of |null − chance| over 200 permutations (--calibrate-null, before this registration).",

        ("A2", "the null lane's agreement lies within null_band 0.13 of chance, the 0.99 quantile of |null - chance| over 200 permutations"),
    ):

        if not pat.search(text):
            raise SystemExit(f"{pid}: sentence not found")
        text = pat.sub(lambda m, s=sentence: m.group(1) + '"' + s + '"', text, count=1)
    text = text.replace("constants = { margin = 0.25 }", "constants = { margin = 0.15 }", 1)
    text = text.replace("constants = { null_band = 0.1 }", "constants = { null_band = 0.13 }", 1)

    text = readme.read_text(encoding="utf-8")
    text = re.sub(r"\*\*The one question\.\*\* <[^>]*>",

    text = re.sub(r"\*\*What `evaluate_rows\(\)` returns\.\*\* <[^>]*>\.",
                  "**What `evaluate_rows()` returns.** [A1] `a1_pass` (every labelled person\nflagged), [A2] `a2_pass` (the null lane within null_band of chance), then\n[C1] `c1_pass` (agreement at least the null lane's plus margin); when [A1] or\n[A2] fails, `c1_pass` is None — no verdict.", text, flags=re.S)
    readme.write_text(text, encoding="utf-8")

    sh([py, str(prep)], cwd=probe)
    # the band on the correct model, as the author derived it before the registration
    correct = measure.read_text(encoding="utf-8")
    measure.write_text(correct.replace("/ sd**2", "/ sd"), encoding="utf-8")
    calib = sh([py, str(measure), "--calibrate-null"], cwd=probe)
    measure.write_text(correct, encoding="utf-8")
    band = float(re.search(r"null_band by the rule[^:]*: ([0-9.]+)", calib).group(1))
    if not band <= 0.13:
        raise SystemExit(f"calibrated band {band} is above the registered 0.13")
    sh(["git", "add", "-A", "experiments"], cwd=probe)
    sh(["git", "commit", "-q", "-m", "Register apply-screen: agreement with the clinicians against the permuted null, the scripts and the prepared persons"], cwd=probe, **GIT_ENV)

    sheet = probe / "TASK.md"
    text = sheet.read_text(encoding="utf-8")
    start = text.index("Run your experiment through labkit as a protocol: the area")
    end = text.index("lands in the run's `metrics.json` under the key `c1_score`;")
    text = text[:start] + """You continue an experiment someone started; it runs through labkit as a
protocol. The area `experiments/01-apply-screen/` holds it: its protocol is
registered and committed (repeat cost low, kind apply) — question,
criterion, admissibility condition and null condition with their
constants —, with the scripts that prepare the persons and apply the
model, and the prepared data. It was measured once, after the
registration (the run under `runs/`, committed). What is owed now is what
follows a first measurement: its analysis audit, the decisions on what the
audit finds, and a number the protocol and REPORT.md can carry. The kept number
""" + text[end:]
    sheet.write_text(text, encoding="utf-8")
    sh(["git", "add", "TASK.md"], cwd=probe)
    sh(["git", "commit", "-q", "-m", "Probe: the task continues a registered experiment"], cwd=probe, **GIT_ENV)
    print(f"built: {probe} (labkit {sha}, calibrated band {band})")


if __name__ == "__main__":
    main()
