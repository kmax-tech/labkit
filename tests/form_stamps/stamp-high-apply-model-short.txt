`protocol-fill PROTOCOL-<name>.md "<label>=<value>"`.*

```toml
# `[result] run = "<run_id>"` below [protocol] is written by `labkit-adopt`:
# the one declaration that this protocol reports a number.
[protocol]
standard = "0.31"
id = "p-0000000000000000"
type = "<exploratory | confirmatory>"
repeat_cost = "high"
kind = "apply"
frozen = false
preflight_audit = ""

[kept]
key = "accuracy"
unit = "1"

[model]
alias = "haiku"
# `[kept]`: the key and unit of the number this protocol reports, at every kind; a
# criterion judges that number. With a model: `[model] alias` (stamped by --model),
# `[model.arms.<arm>]` with profile, fingerprint, alias (finalize holds them to the run); `[gate] labels = […]`.

# The promises as data, one table per id: the key the run reports, the
# sentence (no `<…>` left at the freeze), the code region, and the constants
# the script reads with promises.all_constants(). Each constant stands in one
# table. The numbers below are the stamp's examples, not your design: set
# each before the commit to the value its sentence states, chosen from the
# question, never from a result. At an apply with the estimator skeleton,
# the null condition registers the rule, not the value: each run computes its
# permitted deviation (`--calibrate-null` previews it). After the first
# measurement none of them changes.
# The criterion — the bar the kept number is judged against: optional
# while type = "exploratory", owed when "confirmatory" (or undecided).
# Two steps, both owed: (1) uncomment the table below and name its id,
# in brackets, in the Inference criterion field; (2) uncomment the
# block between the criterion-1 markers in evaluate_rows of
# scripts/01_measure.py, which reports criterion_1_threshold and
# criterion_1_pass. One step alone is held (code-check, intent-check).
# [[promise]]
# id = "criterion-1"
# key = "criterion_1_pass"
# sentence = "<the criterion in one sentence: registered accuracy >= null accuracy + margin — or, for a bar the task states, >= at_least, its source named (constants = { at_least = … } in place of margin); accuracy over all items of the lane, an empty answer counts as not correct>"
# scope = ["scripts/01_measure.py"]
# constants = { margin = 0.25 }

[[promise]]
```
